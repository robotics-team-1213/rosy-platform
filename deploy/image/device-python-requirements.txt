# The rest is their closure as resolved on 2026-09-24 for CPython 3.12.

# one hash. Regenerate for BOTH platforms and update inputs.lock.yaml
# python_runtime.requirements_sha256 in the same commit:
#   pip install --dry-run --ignore-installed --report r.json --target t
#     --platform manylinux2014_aarch64 --platform manylinux_2_17_aarch64
#     --platform manylinux_2_28_aarch64 --python-version 3.12
#     --implementation cp --abi cp312 --only-binary=:all: <top-level pins>
#   (again with the x86_64 platform tags), then copy name, version, sha256.
